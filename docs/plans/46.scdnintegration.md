# SCDN Integration for Saddle Robot

> Static Content Delivery Network integration for
> hosting browser assets, capture files, and deployment payloads.

## Overview

The saddle robot integrates with Static Content Delivery
Networks (SCDN) to serve browser assets, capture files,
and deployment payloads from edge locations worldwide.

## SCDN Use Cases

1. Host browser automation assets (scripts, styles,
   configuration files)
2. Serve capture session data (JSON logs, video files)
3. Deliver deployment payloads to deployed sites
4. Cache bot responses for faster repeat execution
5. Distribute browser profiles and extensions

## SCDN Pipeline

```
Capture Pipeline
────────────────────────────────────────────────────────
1. Browser session captures data
2. Assets uploaded to SCDN via HTTPS
3. SCDN caches content at edge locations
4. Bot workers fetch assets from nearest edge
5. Results stored in repository/bucket storage
```

## Implementation

### SCDN Client

```typescript
interface SCDNClientConfig {
  endpoint: string
  token: string
  bucketId: string
  region?: string
  ttlSeconds: number
}

class SCDNClient {
  private config: SCDNClientConfig

  constructor(config: SCDNClientConfig) {
    this.config = config
  }

  async upload(
    path: string,
    content: Buffer | string,
    options?: { contentType?: string; ttl?: number }
  ): Promise<{ url: string; etag: string; size: number }> {
    try {
      const response = await fetch(
        `${this.config.endpoint}/${this.config.bucketId}/${path}`,
        {
          method: 'PUT',
          headers: {
            Authorization: `Bearer ${this.config.token}`,
            'Content-Type': options?.contentType ?? 'application/octet-stream',
            'Cache-Control': `max-age=${options?.ttl ?? this.config.ttlSeconds}`,
          },
          body:
            typeof content === 'string'
              ? new TextEncoder().encode(content)
              : content,
        }
      )

      if (!response.ok) {
        throw new Error(
          `SCDN upload failed: ${response.status} ${response.statusText}`
        )
      }

      const etag = response.headers.get('etag') ?? ''
      const url = `${this.config.endpoint}/${this.config.bucketId}/${path}`
      const size =
        typeof content === 'string'
          ? new TextEncoder().encode(content).byteLength
          : content.byteLength

      return { url, etag, size }
    } catch (error) {
      throw new Error(
        `SCDNClient.upload("${path}") failed: ${error instanceof Error ? error.message : String(error)}`
      )
    }
  }

  async download(path: string): Promise<Buffer> {
    try {
      const response = await fetch(
        `${this.config.endpoint}/${this.config.bucketId}/${path}`,
        {
          headers: {
            Authorization: `Bearer ${this.config.token}`,
          },
        }
      )

      if (!response.ok) {
        throw new Error(
          `SCDN download failed: ${response.status} ${response.statusText}`
        )
      }

      const arrayBuffer = await response.arrayBuffer()
      return Buffer.from(arrayBuffer)
    } catch (error) {
      throw new Error(
        `SCDNClient.download("${path}") failed: ${error instanceof Error ? error.message : String(error)}`
      )
    }
  }

  async delete(path: string): Promise<boolean> {
    try {
      const response = await fetch(
        `${this.config.endpoint}/${this.config.bucketId}/${path}`,
        {
          method: 'DELETE',
          headers: {
            Authorization: `Bearer ${this.config.token}`,
          },
        }
      )
      return response.ok
    } catch (error) {
      throw new Error(
        `SCDNClient.delete("${path}") failed: ${error instanceof Error ? error.message : String(error)}`
      )
    }
  }
}
```

### Asset Upload Flow

```typescript
// In browser capture module
import { SCDNClient } from '@devthink/saddle/deploy'

async function uploadCaptureSession(
  sessionData: SessionData,
  scdnClient: SCDNClient
): Promise<SessionResult> {
  const sessionId = generateSessionId()
  const sessionPath = `sessions/${sessionId}.json`
  const screenshotPath = `screenshots/${sessionId}.png`

  // Upload session data
  const sessionResult = await scdnClient.upload(
    sessionPath,
    JSON.stringify(sessionData),
    { contentType: 'application/json', ttl: 3600 }
  )

  // Upload screenshot
  const screenshotResult = await scdnClient.upload(
    screenshotPath,
    sessionData.screenshot,
    { contentType: 'image/png', ttl: 86400 }
  )

  return {
    sessionId,
    sessionUrl: sessionResult.url,
    screenshotUrl: screenshotResult.url,
  }
}
```

## Configuration

### Environment Variables

| Variable | Description | Required |
|----------|-------------|----------|
| `SBOT_SCDN_ENDPOINT` | SCDN API endpoint | Yes |
| `SBOT_SCDN_TOKEN` | SCDN authentication token | Yes |
| `SBOT_SCDN_BUCKET` | Bucket ID for storage | Yes |
| `SBOT_SCDN_REGION` | Preferred edge region | No |
| `SBOT_SCDN_TTL` | Default cache TTL in seconds | No |

### Example Configuration

```yaml
# .saddle/scdn.yml
scdn:
  endpoint: "https://api.scdn.example.com/v1"
  bucket: "saddle-assets"
  token: "${SBOT_SCDN_TOKEN}"
  region: "us-east-1"
  ttl: 3600
  retry:
    maxAttempts: 3
    delayMs: 1000
```

## Error Handling

```typescript
interface SCDNResult {
  success: boolean
  url?: string
  error?: string
  retryCount?: number
}

async function safeUpload(
  client: SCDNClient,
  path: string,
  content: Buffer,
  maxRetries: number = 3
): Promise<SCDNResult> {
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      const result = await client.upload(path, content)
      return { success: true, url: result.url }
    } catch (error) {
      if (attempt === maxRetries) {
        return {
          success: false,
          error: error instanceof Error ? error.message : String(error),
          retryCount: attempt,
        }
      }
      await new Promise((resolve) =>
        setTimeout(resolve, 1000 * attempt)
      )
    }
  }
  return { success: false, error: 'Unexpected error' }
}
```
